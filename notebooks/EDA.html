from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report
import time as t

n_estimators=300,
class_weight='balanced',
random_state=42,
n_jobs =- 1
)
start_time = t.time()
rf. fit(X_train, y_train)
end_time = t.time()
